import java.io.BufferedReader;
import java.io.FileReader;
import java.io.FileWriter;
import java.util.HashMap;
import java.util.Vector;

class MNTEntry {
    String name;
    int pp, kp, mdtp, kpdtp;

    public MNTEntry(String name, int pp, int kp, int mdtp, int kpdtp) {
        this.name = name;
        this.pp = pp;
        this.kp = kp;
        this.mdtp = mdtp;
        this.kpdtp = kpdtp;
    }

    public int getPp() { return pp; }
    public int getKp() { return kp; }
    public int getMdtp() { return mdtp; }
    public int getKpdtp() { return kpdtp; }
}

public class MacroPass2 {

    public static void main(String[] args) throws Exception {

        BufferedReader irb = new BufferedReader(new FileReader("intermediate.txt"));
        BufferedReader mdtb = new BufferedReader(new FileReader("mdt.txt"));
        BufferedReader kpdtb = new BufferedReader(new FileReader("kpdt.txt"));
        BufferedReader mntb = new BufferedReader(new FileReader("mnt.txt"));

        FileWriter fr = new FileWriter("pass2.txt");

        HashMap<String, MNTEntry> mnt = new HashMap<>();
        HashMap<Integer, String> aptab = new HashMap<>();
        HashMap<String, Integer> aptabInverse = new HashMap<>();

        Vector<String> mdt = new Vector<>();
        Vector<String> kpdt = new Vector<>();

        String line;

        /* READ MDT */
        while ((line = mdtb.readLine()) != null)
            mdt.add(line);

        /* READ KPDT */
        while ((line = kpdtb.readLine()) != null)
            kpdt.add(line);

        /* READ MNT */
        while ((line = mntb.readLine()) != null) {
            String parts[] = line.split("\\s+");
            mnt.put(parts[0],
                    new MNTEntry(parts[0],
                            Integer.parseInt(parts[1]),
                            Integer.parseInt(parts[2]),
                            Integer.parseInt(parts[3]),
                            Integer.parseInt(parts[4])));
        }

        /* PROCESS INTERMEDIATE CODE */
        while ((line = irb.readLine()) != null) {

            String parts[] = line.split("\\s+");

            /* Macro Call Detected */
            if (mnt.containsKey(parts[0])) {

                int pp = mnt.get(parts[0]).getPp();
                int kp = mnt.get(parts[0]).getKp();
                int mdtp = mnt.get(parts[0]).getMdtp();
                int kpdtp = mnt.get(parts[0]).getKpdtp();

                int paramNo = 1;

                /* Positional parameters */
                for (int i = 0; i < pp; i++) {
                    parts[paramNo] = parts[paramNo].replace(",", "");
                    aptab.put(paramNo, parts[paramNo]);
                    aptabInverse.put(parts[paramNo], paramNo);
                    paramNo++;
                }

                /* Keyword parameters */
                int j = kpdtp - 1;
                for (int i = 0; i < kp; i++) {
                    String temp[] = kpdt.get(j).split("\t");
                    aptab.put(paramNo, temp[1]);  // default value
                    aptabInverse.put(temp[0], paramNo);
                    j++;
                    paramNo++;
                }

                /* Override keyword values from macro call */
                for (int i = pp + 1; i < parts.length; i++) {
                    parts[i] = parts[i].replace(",", "");
                    String splits[] = parts[i].split("=");
                    String name = splits[0].replace("&", "");
                    aptab.put(aptabInverse.get(name), splits[1]);
                }

                /* EXPANSION OF MACRO BODY */
                int index = mdtp - 1;

                while (!mdt.get(index).equalsIgnoreCase("MEND")) {

                    String splits[] = mdt.get(index).split("\\s+");
                    fr.write("+");

                    for (int k = 0; k < splits.length; k++) {

                        if (splits[k].contains("(P,")) {
                            splits[k] = splits[k].replaceAll("[^0-9]", "");
                            String value = aptab.get(Integer.parseInt(splits[k]));
                            fr.write(value + "\t");
                        } else {
                            fr.write(splits[k] + "\t");
                        }
                    }

                    fr.write("\n");
                    index++;
                }

                aptab.clear();
                aptabInverse.clear();
            }

            /* Ordinary instruction */
            else {
                fr.write(line + "\n");
            }
        }

        fr.close();
        irb.close();
        mdtb.close();
        kpdtb.close();
        mntb.close();
    }
}

/***************************************************************************************
   THEORY: MACRO PROCESSOR – PASS 2
****************************************************************************************

AIM:
To implement Pass 2 of a Macro Processor that performs macro expansion,
substitutes actual parameters, handles keyword/positional parameters,
and produces the final expanded assembly program.

INTRODUCTION:
A macro processor allows the programmer to define macros that act as
shorthand notations for commonly used instruction sequences.
A two-pass macro processor consists of:
   • PASS 1 → Macro Definition Processing
   • PASS 2 → Macro Expansion

WHAT PASS 2 DOES:
✓ Reads the **Intermediate Code** produced by PASS 1  
✓ Identifies macro calls  
✓ Creates **APTAB (Argument Parameter Table)**  
✓ Replaces each (P, n) from MDT with actual parameters  
✓ Writes fully expanded code to **pass2.txt**  
✓ Expands macros using:
      - MNT (Macro Name Table)
      - MDT (Macro Definition Table)
      - KPDTAB (Keyword Parameter Default Table)

IMPORTANT DATA STRUCTURES:

1) **MNT (Macro Name Table)**  
   Stores:
      - Macro name  
      - Number of Positional Parameters (PP)  
      - Number of Keyword Parameters (KP)  
      - Pointer to MDT (MDTP)  
      - Pointer to KPDTAB (KPDTABP)  

2) **MDT (Macro Definition Table)**  
   Contains the actual body of each macro, including (P, n) references.

3) **KPDTAB (Keyword Parameter Default Table)**  
   Stores:
      - Keyword parameter name  
      - Its default value  

4) **APTAB (Argument Parameter Table)**  
   Mapping of parameter index → argument value  
   (Built fresh for each macro call)

HOW PASS 2 WORKS:
1) Read each line of intermediate file.
2) If the line is a macro call:
      • Retrieve PP, KP, MDTP, KPDTABP from MNT.  
      • Build APTAB for positional parameters.  
      • Insert default values of keyword parameters from KPDTAB.  
      • Override defaults using values from macro call.  
3) Expand Macro:
      • Fetch each line from MDT using MDTP.  
      • Replace (P,n) with APTAB[n].  
      • Stop when "MEND" is reached.  
4) Write all expanded lines to pass2.txt.

ADVANTAGES:
✓ Reduces code repetition  
✓ Makes programs modular and readable  
✓ Handles positional and keyword parameters  
✓ Produces fully expanded assembly program  
✓ Supports nested and repeated macro calls  

CONCLUSION:
Pass 2 of the macro processor performs complete macro expansion
by substituting parameters and generating executable assembly code.
It uses MNT, MDT, KPDTAB, and APTAB to accurately replace all macro
instructions with their expanded forms.

****************************************************************************************/
