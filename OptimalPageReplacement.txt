/* ============================================================
                  OPTIMAL PAGE REPLACEMENT
   ============================================================ */

import java.util.Scanner;

public class OptimalPageReplacement {

    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        int noOfPages, capacity, ptr = 0, hit = 0, fault = 0;
        boolean isFull = false;
        double hitRatio, faultRatio;

        System.out.print("Enter number of pages: ");
        noOfPages = sc.nextInt();

        int pages[] = new int[noOfPages];
        System.out.println("Enter page reference string: ");
        for (int i = 0; i < noOfPages; i++) {
            pages[i] = sc.nextInt();
        }

        System.out.print("Enter frame capacity: ");
        capacity = sc.nextInt();

        int frame[] = new int[capacity];
        int table[][] = new int[noOfPages][capacity];

        for (int i = 0; i < capacity; i++) {
            frame[i] = -1;
        }

        System.out.println("---------------------------------------------------------------------");

        for (int i = 0; i < noOfPages; i++) {

            int search = -1;

            // Check for page hit
            for (int j = 0; j < capacity; j++) {
                if (frame[j] == pages[i]) {
                    search = j;
                    hit++;
                    System.out.printf("%4s", "H");
                    break;
                }
            }

            // Page fault
            if (search == -1) {

                if (isFull) {

                    int[] index = new int[capacity];
                    boolean[] indexFlag = new boolean[capacity];

                    // Find next use of every frame
                    for (int j = i + 1; j < noOfPages; j++) {
                        for (int k = 0; k < capacity; k++) {
                            if ((pages[j] == frame[k]) && (!indexFlag[k])) {
                                index[k] = j;
                                indexFlag[k] = true;
                                break;
                            }
                        }
                    }

                    // Find page with farthest next use
                    int max = index[0];
                    ptr = 0;

                    if (max == 0) max = 9999;

                    for (int j = 0; j < capacity; j++) {
                        if (index[j] == 0) index[j] = 9999;

                        if (index[j] > max) {
                            max = index[j];
                            ptr = j;
                        }
                    }
                }

                frame[ptr] = pages[i];
                fault++;
                System.out.printf("%4s", "F");

                if (!isFull) {
                    ptr++;
                    if (ptr == capacity) {
                        ptr = 0;
                        isFull = true;
                    }
                }
            }

            System.arraycopy(frame, 0, table[i], 0, capacity);
        }

        System.out.println("\n---------------------------------------------------------------------");

        // Print page table
        for (int i = 0; i < capacity; i++) {
            for (int j = 0; j < noOfPages; j++) {
                System.out.printf("%4d", table[j][i]);
            }
            System.out.println();
        }

        System.out.println("---------------------------------------------------------------------");

        hitRatio = ((double) hit / noOfPages) * 100;
        faultRatio = ((double) fault / noOfPages) * 100;

        System.out.println("Total Page Faults: " + fault);
        System.out.println("Total Page Hits:   " + hit);
        System.out.printf("Hit Ratio: %.2f%%\n", hitRatio);
        System.out.printf("Fault Ratio: %.2f%%\n", faultRatio);

        sc.close();
    }
}



/* =========================================================================
   THEORY: OPTIMAL PAGE REPLACEMENT ALGORITHM
   =========================================================================

   AIM:
   To implement the Optimal Page Replacement algorithm and compute
   page hits, page faults, hit ratio, and fault ratio.

   INTRODUCTION:
   When a process executes, it frequently accesses memory pages.
   Due to limited frames in main memory, a page replacement
   algorithm decides which page to remove when a new page must be loaded.

   Optimal Page Replacement replaces the page that will NOT be used
   for the longest period of time in the future.

   It gives the *least possible* page faults but cannot be implemented
   in real operating systems (because it needs future knowledge).

   DEFINITIONS:
   ● Page Hit  → page already present in a frame  
   ● Page Fault → page not found → must be loaded  
   ● Frame → fixed-size block of physical memory  
   ● Reference String → sequence of page requests  

   WORKING OF OPTIMAL ALGORITHM:
   1. If free frames exist: load the page directly.
   2. If the page exists in memory → HIT.
   3. If page is not in memory → FAULT:
        • For each page in frame, check its next future use.
        • The page which is not used for the longest time OR not used again at all
          is replaced.
   4. Repeat for all page references.

   KEY ADVANTAGE:
   ✓ Produces the minimum number of page faults possible.
   ✓ Ideal reference algorithm for comparison (benchmark).

   DISADVANTAGES:
   ✗ Requires future knowledge of reference string → not possible in real OS.  
   ✗ Only used for theoretical analysis.

   APPLICATIONS:
   • Performance comparison with FIFO, LRU, LFU, etc.  
   • Benchmarking memory management algorithms  

   CONCLUSION:
   Optimal Page Replacement is the best theoretical page replacement algorithm.
   It always leads to minimum page faults but is impractical for real systems.
   It is mainly used as a standard to evaluate other algorithms.
   ========================================================================= */
