/* ============================================================
                     FIFO PAGE REPLACEMENT
   ============================================================ */

import java.util.Scanner;

public class FIFO {

    public static void main(String args[]) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter the number of pages: ");
        int noOfPages = sc.nextInt();

        int pages[] = new int[noOfPages];
        System.out.println("Enter the page reference string: ");
        for (int i = 0; i < noOfPages; i++)
            pages[i] = sc.nextInt();

        System.out.print("Enter the capacity of frames: ");
        int capacity = sc.nextInt();

        int frame[] = new int[capacity];
        int table[][] = new int[noOfPages][capacity];
        int index = 0, hit = 0, fault = 0;

        // Initialize all frames as empty (-1)
        for (int i = 0; i < capacity; i++)
            frame[i] = -1;

        System.out.println("\n-----------------------------------------------------------");

        for (int i = 0; i < noOfPages; i++) {

            int search = -1;

            // Check for page hit
            for (int j = 0; j < capacity; j++) {
                if (frame[j] == pages[i]) {
                    search = j;
                    hit++;
                    System.out.printf("%4s", "H");
                    break;
                }
            }

            // Page fault condition
            if (search == -1) {
                frame[index] = pages[i];
                fault++;
                System.out.printf("%4s", "F");

                index++;
                if (index == capacity)
                    index = 0; // FIFO replacement
            }

            // Copy current frame state into table
            System.arraycopy(frame, 0, table[i], 0, capacity);
        }

        System.out.println("\n-----------------------------------------------------------");

        // Print page replacement table
        for (int i = 0; i < capacity; i++) {
            for (int j = 0; j < noOfPages; j++)
                System.out.printf("%3d ", table[j][i]);
            System.out.println();
        }

        System.out.println("-----------------------------------------------------------");

        double hitRatio = (double) hit / noOfPages * 100;
        double faultRatio = (double) fault / noOfPages * 100;

        System.out.println("Total Page Faults: " + fault);
        System.out.println("Total Page Hits:   " + hit);
        System.out.printf("Hit Ratio:  %.2f%%\n", hitRatio);
        System.out.printf("Fault Ratio: %.2f%%\n", faultRatio);

        sc.close();
    }
}


/* ========================================================================
   THEORY: FIFO (FIRST IN FIRST OUT) PAGE REPLACEMENT ALGORITHM
   ========================================================================

   AIM:
   To implement the FIFO Page Replacement Algorithm and calculate
   page hits, page faults, hit ratio, and fault ratio.

   INTRODUCTION:
   In operating systems, processes request memory pages.  
   When memory frames are full, a **page replacement algorithm** selects 
   which page to remove.

   FIFO is the simplest algorithm.  
   It removes the page that entered memory **first (oldest page)**.

   ------------------------------------------------------------------------
   HOW FIFO WORKS:
   ------------------------------------------------------------------------
   1. Pages are inserted in the order they arrive.
   2. On page fault:
         • If frame has space → insert page
         • If frame is full → remove the oldest page
   3. Use a pointer/index to track which page came first.
   4. Display hit (“H”) if page exists already, else fault (“F”).

   ------------------------------------------------------------------------
   TERMINOLOGY:
   • **Page Hit** → Requested page is already in a frame  
   • **Page Fault** → Page is not in memory → must be loaded  
   • **Frame** → Fixed-size block of RAM  
   • **Reference String** → Sequence of requested pages  

   ------------------------------------------------------------------------
   ADVANTAGES:
   ✓ Simple and easy to implement  
   ✓ Requires very little overhead  
   ✓ Uses queue-like mechanism  

   ------------------------------------------------------------------------
   DISADVANTAGES:
   ✗ May evict important pages unnecessarily  
   ✗ High page faults in some patterns (Belady’s Anomaly)  
   ✗ No priority to frequently used pages  

   ------------------------------------------------------------------------
   APPLICATIONS:
   • Basic memory management  
   • Benchmarking against other algorithms (LRU, Optimal)  
   • Educational understanding of paging  

   ------------------------------------------------------------------------
   CONCLUSION:
   FIFO Page Replacement replaces the oldest loaded page on a fault.
   Although simple, it is not optimal and may behave poorly in some cases.
   It is useful as a baseline algorithm to compare other techniques.
   ======================================================================== */
